# 1. 切换目标框架（仓库默认 net6.0 适配沙箱）
#    编辑 src/MCLCS.App/MCLCS.App.csproj，确认 <TargetFramework>net8.0-windows</TargetFramework>

# 2. 编译
dotnet restore MCLCS.sln
dotnet build MCLCS.sln -c Release

# 3. 发布单文件 EXE
dotnet publish src/MCLCS.App -c Release -r win-x64 --self-contained true -p:PublishSingleFile=true

# 产物在 src/MCLCS.App/bin/Release/net8.0-windows/win-x64/publish/MCLCS.App.exe
